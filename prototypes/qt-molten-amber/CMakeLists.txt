cmake_minimum_required(VERSION 3.21)
project(Phosphor002 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Quick)
add_library(molten renderer.cpp)
target_link_libraries(molten PUBLIC Qt6::Gui)
target_include_directories(molten PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(molten PRIVATE -Wall -Wextra -Wpedantic)
add_executable(phosphor-002 main.cpp)
target_link_libraries(phosphor-002 PRIVATE molten Qt6::Quick)
include(CTest)
if(BUILD_TESTING)
  add_executable(renderer-tests tests.cpp)
  target_link_libraries(renderer-tests PRIVATE molten)
  add_test(NAME glyph-material COMMAND renderer-tests)
  set_tests_properties(glyph-material PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
