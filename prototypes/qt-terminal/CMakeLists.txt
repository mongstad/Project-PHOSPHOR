cmake_minimum_required(VERSION 3.21)
project(PhosphorTerminal LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Quick ShaderTools)
# Prefer the distribution's libvterm. A local upstream source tree is an
# explicit build-only fallback; nothing is fetched or installed by CMake.
set(LIBVTERM_SOURCE_DIR "" CACHE PATH "Optional extracted upstream libvterm 0.3.3 source")
if(LIBVTERM_SOURCE_DIR)
 file(GLOB VTERM_SOURCES CONFIGURE_DEPENDS "${LIBVTERM_SOURCE_DIR}/src/*.c")
 add_library(vterm STATIC ${VTERM_SOURCES})
 target_include_directories(vterm PUBLIC "${LIBVTERM_SOURCE_DIR}/include")
else()
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(VTERM REQUIRED IMPORTED_TARGET vterm>=0.3.3)
 add_library(vterm ALIAS PkgConfig::VTERM)
endif()
add_library(terminal-core terminal.cpp pty.cpp)
target_link_libraries(terminal-core PUBLIC Qt6::Gui vterm)
target_include_directories(terminal-core PUBLIC .)
add_library(terminal-view view.cpp ../qt-molten-amber/renderer.cpp ../qt-gpu-atlas/fixture.cpp)
target_include_directories(terminal-view PUBLIC ../qt-molten-amber ../qt-gpu-atlas)
target_link_libraries(terminal-view PUBLIC terminal-core Qt6::Quick)
qt_add_shaders(terminal-view "glyph_shaders" PREFIX "/shaders" BATCHABLE FILES glyph.vert glyph.frag)
add_executable(phosphor-terminal main.cpp)
target_link_libraries(phosphor-terminal PRIVATE terminal-view)
foreach(target terminal-core terminal-view phosphor-terminal)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
 add_executable(terminal-tests tests.cpp)
 target_link_libraries(terminal-tests PRIVATE terminal-view)
 add_test(NAME terminal-core-pty COMMAND terminal-tests)
 set_tests_properties(terminal-core-pty PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_FORCE_STDERR_LOGGING=1" TIMEOUT 30)
endif()
