cmake_minimum_required(VERSION 3.21)
project(PhosphorGpuAtlas LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Core Gui Quick)
# Reuse the accepted material verbatim, including its known compiler limitation.
add_library(glyph-data ../qt-molten-amber/renderer.cpp fixture.cpp atlas.cpp)
target_include_directories(glyph-data PUBLIC . ../qt-molten-amber)
target_link_libraries(glyph-data PUBLIC Qt6::Gui)
add_executable(phosphor-003 main.cpp view.cpp)
target_link_libraries(phosphor-003 PRIVATE glyph-data Qt6::Quick)
target_compile_options(phosphor-003 PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(atlas-tests tests.cpp)
 target_link_libraries(atlas-tests PRIVATE glyph-data)
 add_test(NAME glyph-atlas COMMAND atlas-tests)
 set_tests_properties(glyph-atlas PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_FORCE_STDERR_LOGGING=1")
endif()
